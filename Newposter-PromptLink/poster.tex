  % ---------------------------------------------------------------------------
  \begin{column}{0.415\textwidth}
    \begin{block}{Why an Open-Weight, Role-Specialized System?}
      \justifying
      \smallcapslabel{Deployment gap.} Closed-model APIs perform well on EHR
      prediction but raise privacy and compliance concerns. Local fine-tuning
      avoids the API but needs labeled cohorts, training compute, and
      per-task adaptation.

      \vspace{0.30em}
      \smallcapslabel{Evaluation gap.} Medical multi-agent systems are usually
      scored as one block, so it is unclear which role controls the prediction
      and whether retrieval drives the gain.

      \vspace{0.45em}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.45em]{notebox}
        \textbf{Q1.} Which role sets the operating point at which patients are
        flagged?\\[0.25em]
        \textbf{Q2.} Can open-weight agents approach closed-model prompting
        without any training?
      \end{beamercolorbox}

      \vspace{0.45em}
      \smallcapslabel{Design (Figure 1).}
      \begin{itemize}\tightitems
        \item \textcolor{AnalystBlue}{\textbf{Two label-blind analysts.}}
          The risk analyst reads the target record with the nearest
          positive-outcome exemplar; the protective analyst reads it with the
          nearest negative-outcome exemplar. Neither sees a label or predicts.
        \item \textcolor{VTMaroon}{\textbf{One integrator.}} Reconciles both
          readings, issues one focused knowledge query, and outputs the risk
          score $\hat p$; $\hat y = \mathbf{1}[\hat p > 0.5]$.
        \item \textcolor{VTOrange}{\textbf{Controlled retrieval.}} MedCPT
          retrieval over MedCorp\,+\,UMLS and KARE patient exemplars are
          fixed; only topology and model placement vary.
      \end{itemize}

      \vspace{0.35em}
      \smallcapslabel{Models.} Large $=$ GPT-OSS-120B, small $=$ Qwen2.5-7B,
      both served locally. Reported system: large analysts, small integrator.
    \end{block}

    \vspace{0.25cm}

    \begin{block}{What Carries the Gain? Topology, Not Retrieval}
      \begin{columns}[T,totalwidth=\linewidth]
        \begin{column}{0.56\linewidth}
          \centering
          \textbf{GPT-OSS-120B, MIMIC-IV mortality}\\[0.15em]
          \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.56\linewidth}cc@{}}
            \toprule
            \textbf{Configuration} & \textbf{F1} & \textbf{Sens.} \\
            \midrule
            CoT (no retrieval)        & 58.3 & 24.2 \\
            single $+$ RAG            & 37.7 & 17.9 \\
            single $+$ RAG $+$ Sim    & 52.8 & 22.1 \\
            multi $+$ Sim             & 51.3 & 10.0 \\
            multi $+$ RAG $+$ Sim     & 54.4 & 14.7 \\
            \bottomrule
          \end{tabular}
        \end{column}
        \begin{column}{0.42\linewidth}
          \justifying
          Retrieved knowledge alone \emph{hurts} a single agent
          (58.3 $\to$ 37.7). The exemplar pair helps once it is split across
          two contrastive analysts, and removing MedRAG from the pipeline moves
          F1 by only 3.1 points.
        \end{column}
      \end{columns}
    \end{block}
  \end{column}

  % ---------------------------------------------------------------------------
  \begin{column}{0.575\textwidth}
    \begin{block}{Pipeline: Multi-Source Databases and Role-Specialized Retrieval}
      \centering
      \includegraphics[width=\linewidth]{Newposter-PromptLink/figures/Paper/main.pdf}

      \vspace{0.25em}
      \justifying
      \textbf{Figure 1.} \textbf{(A)} A knowledge database is built from
      MedCorp and UMLS, and a patient database from the EHRs. \textbf{(B)} The
      label-blind risk and protective analysts read the target record and
      retrieve from both databases; the integrator runs its own retrieval and
      integrates the two readings into the final prediction.
    \end{block}
  \end{column}

\end{columns}

\begin{column}{0.315\textwidth}

  \begin{block}{Q1: The Integrator Sets the Operating Point}
    \centering
    \textbf{MIMIC-IV mortality, fixed $n=100$ subset}

    \vspace{0.20em}
    \begin{tabular}{@{}lccc@{}}
      \toprule
      \textbf{Analyst / Retrieval / Integrator} & \textbf{F1} & \textbf{Sens.} & \textbf{AUROC} \\
      \midrule
      oss / oss / oss   & 38.5 & 13.0 & .543 \\
      oss / qwen / oss  & 35.0 &  7.4 & .533 \\
      qwen / oss / oss  & 36.3 & 11.1 & .517 \\
      \rowcolor{IntegratorGold!24}
      \textbf{oss / oss / qwen} & \textbf{57.4} & \textbf{64.8} & \textbf{.603} \\
      qwen / qwen / qwen & 54.6 & 59.3 & .491 \\
      oss CoT (single) & 40.9 & 22.2 & .536 \\
      \bottomrule
    \end{tabular}

    \vspace{0.30em}
    \justifying
    Swapping only the analyst or only the retrieval model changes little.
    Swapping only the integrator lifts sensitivity from 13.0 to 64.8.
    The same lift reproduces with GPT-4o analysts: a Qwen integrator raises
    recall from 3.7 to 57.4 (MIMIC-III mortality, $n=100$).
  \end{block}

  \vspace{0.25cm}

  \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.55em]{resultbox}
    \textcolor{VTMaroon}{\textbf{Takeaways}}\par\vspace{0.15em}
    \begin{enumerate}\tightitems
      \item \textbf{Role placement matters:} the integrator, not the analysts
        or the retriever, sets the high-recall operating point.
      \item \textbf{Topology matters:} contrastive role separation makes
        exemplars and retrieved knowledge usable; retrieval alone does not.
      \item \textbf{Task signal bounds the gain:} mortality benefits;
        readmission exposes the limit of the available record.
    \end{enumerate}
  \end{beamercolorbox}

\end{column}

% -----------------------------------------------------------------------------
% MIDDLE: main result
% -----------------------------------------------------------------------------
\begin{column}{0.345\textwidth}

  \begin{block}{Q2: Main Result on MIMIC-IV Mortality}
    \centering
    \textbf{Full test set ($\approx$1,000 patients, 19.2\% positive),
    threshold 0.5}

    \vspace{0.20em}
    \rowcolors{2}{white}{SoftGray!70}
    \begin{tabular}{@{}>{\raggedright\arraybackslash}p{0.50\linewidth}ccc@{}}
      \toprule
      \textbf{Method} & \textbf{Acc.} & \textbf{F1} & \textbf{Sens.} \\
      \midrule
      All-negative                       & 80.8 & 44.7 &  0.0 \\
      \multicolumn{4}{@{}l}{\emph{Task-supervised / fine-tuned}} \\
      GraphCare                          & 91.5 & 80.3 & 57.8 \\
      KARE (fine-tuned LLM)              & 94.1 & 90.4 & 73.2 \\
      \multicolumn{4}{@{}l}{\emph{Training-free, closed API}} \\
      Claude 3.5 few-shot                & 84.5 & 57.4 & 15.8 \\
      \multicolumn{4}{@{}l}{\emph{Training-free, open-weight, local (ours)}} \\
      GPT-OSS single $+$ RAG $+$ Sim     & 71.6 & 52.8 & 22.1 \\
      All-large (oss / oss / oss)        & 78.7 & 54.4 & 14.7 \\
      All-small (qwen / qwen / qwen)     & 65.6 & 58.0 & 60.0 \\
      \rowcolor{IntegratorGold!24}
      \textbf{Large analysts, small integrator} & \textbf{66.4} &
      \textbf{58.8} & \textbf{61.1} \\
      \bottomrule
    \end{tabular}

    \vspace{0.45em}
    \begin{columns}[T,totalwidth=\linewidth]
      \begin{column}{0.485\linewidth}
        \bignumber{+1.4 F1}{vs.\ Claude 3.5 few-shot}
      \end{column}
      \begin{column}{0.485\linewidth}
        \bignumber{+45.3 Sens.}{vs.\ Claude 3.5 few-shot}
      \end{column}
    \end{columns}

    \vspace{0.40em}
    \justifying
    The mixed assignment beats GRU, AdaCare, and GraphCare on sensitivity by
    3.3--18.2 points, while those supervised models keep a 17.6--21.5 point
    F1 lead. The all-negative row shows this sensitivity is not majority
    guessing.

    \vspace{0.25em}
    \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.45em]{notebox}
      \centering
      \textbf{Ranking is not improved:} full-set AUROC is 0.763 (single
      large), 0.730 (all-large), 0.706 (mixed). The benefit is a shift of the
      operating point at the default cutoff, not better discrimination.
    \end{beamercolorbox}
  \end{block}

\end{column}

% -----------------------------------------------------------------------------
% RIGHT: mechanism and boundary
% -----------------------------------------------------------------------------
\begin{column}{0.315\textwidth}

  \begin{block}{Mechanism: A Lower Severity Threshold}
    \begin{columns}[T,totalwidth=\linewidth]
      \begin{column}{0.485\linewidth}
        \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.40em]{notebox}
          \centering
          \textcolor{AnalystBlue}{\textbf{Large integrator}}\\[0.10em]
          {\fontsize{20}{24}\selectfont
          Positive only at many recorded conditions\\
          High specificity\\
          Misses moderate-risk deaths}
        \end{beamercolorbox}
      \end{column}
      \begin{column}{0.485\linewidth}
        \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.40em]{resultbox}
          \centering
          \textcolor{VTMaroon}{\textbf{Small integrator}}\\[0.10em]
          {\fontsize{20}{24}\selectfont
          Crosses threshold at fewer conditions\\
          High sensitivity\\
          More false positives}
        \end{beamercolorbox}
      \end{column}
    \end{columns}

    \vspace{0.35em}
    \begin{itemize}\tightitems
      \item Both families rank patients alike by number of recorded
        conditions; they differ in \emph{where} that count becomes a positive
        call.
      \item Changing only the integrator raises the predicted-positive rate by
        \textbf{30.2 points} on MIMIC-IV mortality.
      \item \textbf{56\%} of decedents missed by the large integrator are
        flagged by the small one.
    \end{itemize}
  \end{block}

  \vspace{0.25cm}

  \begin{block}{Boundary: Readmission Is an Honest Negative}
    \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.40em]{boundarybox}
      \centering
      \textbf{Signal in the pre-discharge record (MIMIC-IV)}\\[0.15em]
      \begin{tabular}{@{}lcc@{}}
        & \textbf{Mortality} & \textbf{Readmission}\\
        Bag-of-codes AUROC & 0.77 & 0.59\\
        Single-LLM CoT AUROC & 0.763 & 0.431\\
      \end{tabular}
    \end{beamercolorbox}

    \vspace{0.30em}
    \justifying
    The same lower threshold transfers to 15-day readmission, but the record
    carries little signal for this post-discharge outcome: sensitivity
    $43.5 \to 94.5$, Macro-F1 $47.2 \to 34.7$, specificity $\approx 4\%$.

    \vspace{0.25em}
    \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.40em]{boundarybox}
      \centering\textbf{A sensitive integrator cannot recover outcome signal
      that is absent from the available record.}
    \end{beamercolorbox}
  \end{block}

\end{column}

\end{columns}

      \textcolor{VTMaroon}{\textbf{Scope and limitations}}\par\vspace{0.15em}
      {\fontsize{21}{25}\selectfont
      \begin{columns}[T,totalwidth=0.98\linewidth]
        \begin{column}{0.235\linewidth}
          \textbf{Evidence concentration.} The mixed assignment shows clear
          discrimination only on MIMIC-IV mortality.
        \end{column}
        \begin{column}{0.235\linewidth}
          \textbf{Generalizability.} MIMIC only, two model families, no
          external or prospective validation.
        \end{column}
        \begin{column}{0.235\linewidth}
          \textbf{Not a uniform gain.} Fine-tuned KARE and the single large
          model lead on F1 and AUROC; our edge is the threshold.
        \end{column}
        \begin{column}{0.235\linewidth}
          \textbf{Design and review.} One exemplar per outcome class, no
          top-$k$; no clinician review of intermediate reasoning.
        \end{column}
      \end{columns}}
