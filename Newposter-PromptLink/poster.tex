\setlength{\columnsep}{0.70cm}
\begin{columns}[t,totalwidth=\textwidth]
  \begin{column}{0.415\textwidth}
    \begin{block}{Why Separate the Roles?}
      \justifying
      Clinical prediction from EHRs needs locally deployable models when
      closed-model APIs raise privacy and compliance concerns. Most medical
      multi-agent systems are evaluated as one unit, obscuring which role
      changes the prediction.

      \vspace{0.50em}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.48em]{notebox}
        \textbf{Q1.} Which agent role changes the point at which a patient
        is flagged?\\[0.36em]
        \textbf{Q2.} Can open-weight agents approach closed-model prompting
        without task-specific training?
      \end{beamercolorbox}
    \end{block}

    \vspace{0.28cm}

    \begin{block}{Controlled Role Design}
      \smallcapslabel{Two contrastive analysts}\par
      \textcolor{AnalystBlue}{\textbf{Risk:}} compare the target EHR with a
      positive-outcome exemplar.\\[0.25em]
      \textcolor{ProtectTeal}{\textbf{Protective:}} compare it with a
      negative-outcome exemplar. Neither analyst sees the target label or
      makes the final prediction.

      \vspace{0.42em}
      \smallcapslabel{One integrator}\par
      Reconcile both readings, retrieve additional medical knowledge, and
      output risk $\hat p$; flag when $\hat p>0.5$.

      \vspace{0.44em}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.46em]{resultbox}
        \centering
        \textbf{Studied assignment}\par\vspace{0.08em}
        {\fontsize{26}{30}\selectfont
        GPT-OSS-120B analysts $\rightarrow$ Qwen2.5-7B integrator}
      \end{beamercolorbox}

      \vspace{0.36em}
      \justifying
      MedCorp + UMLS knowledge and the two patient exemplars are fixed
      across role-assignment comparisons.
    \end{block}

    \vspace{0.28cm}

    \begin{block}{Four Evaluation Settings}
      \centering
      {\fontsize{20}{24}\selectfont Positive-outcome prevalence
      (about 1,000 test EHRs per setting)}\par\vspace{0.22em}
      \begin{tabular}{@{}lcc@{}}
        \toprule
        \textbf{Outcome} & \textbf{MIMIC-III} & \textbf{MIMIC-IV} \\
        \midrule
        In-hospital mortality & 5.4\% & \textbf{19.2\%} \\
        15-day readmission & 54.8\% & 46.5\% \\
        \bottomrule
      \end{tabular}

      \vspace{0.28em}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.42em]{notebox}
        \justifying
        \textbf{Why report sensitivity?} On MIMIC-IV mortality, predicting
        every patient as a survivor gives \textbf{80.8\% accuracy} but
        \textbf{0\% sensitivity}.
      \end{beamercolorbox}
    \end{block}
  \end{column}

  \begin{column}{0.575\textwidth}
    \begin{block}{Pipeline: Fixed Retrieval and Specialized Roles}
      \centering
      \includegraphics[width=\linewidth]{Newposter-PromptLink/figures/Paper/main.pdf}
      \vspace{0.24em}
      \justifying
      \textbf{Figure 1.} Medical knowledge and similar patients support two
      contrastive readings. The integrator combines them, runs its own
      focused retrieval, and produces the final risk score.
    \end{block}
  \end{column}
\end{columns}

      \vspace{0.30em}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.46em]{resultbox}
        \centering
        \textbf{Changing only the integrator}\par\vspace{0.12em}
        Macro-F1: \textbf{38.5 $\rightarrow$ 57.4}\quad
        AUROC: \textbf{.543 $\rightarrow$ .603}
      \end{beamercolorbox}

      \vspace{0.28em}
      \justifying
      The analyst or retrieval-model swap does not produce this high-recall
      operating point. These are \textbf{subset results}, separate from the
      full-test results at center.
    \end{block}

    \begin{block}{Component Check: Retrieval and Roles}
      \centering
      {\fontsize{20}{24}\selectfont GPT-OSS-120B, MIMIC-IV mortality;
      macro-F1 (\%)}\par\vspace{0.08em}
      \begin{tikzpicture}[x=0.27cm,y=0.65cm]

        \node[anchor=west,font=\fontsize{19}{22}\selectfont] at (62,5.45) {37.7};

        \node[anchor=west,font=\fontsize{19}{22}\selectfont] at (0,4.0)
          {Single + RAG + exemplars};
        \fill[SoftGray] (0,3.23) rectangle (60,3.67);
        \fill[AnalystBlue] (0,3.23) rectangle (52.8,3.67);

      \vspace{0.10em}
      \justifying
      With the same retrieved inputs, three roles raise macro-F1 from
      \textbf{52.8 to 54.4}. The no-retrieval single model still scores
      \textbf{58.3}; retrieval alone is not the source of the main result.
    \end{block}
  \end{column}

      {\fontsize{20}{24}\selectfont MIMIC-IV, about 1,000 patients;
      19.2\% positive; threshold 0.5}\par\vspace{0.27em}
      {\fontsize{19}{23}\selectfont
      \textcolor{AnalystBlue}{\rule{0.7cm}{0.28cm}}\ Claude 3.5 few-shot\qquad
      \textcolor{VTMaroon}{\rule{0.7cm}{0.28cm}}\ Mixed open-weight system}

      \vspace{0.22em}
      \begin{tikzpicture}[x=0.212cm,y=0.73cm]

        \node[anchor=west,font=\fontsize{19}{22}\selectfont\bfseries]

      \end{tikzpicture}

      \vspace{0.12em}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.48em]{notebox}
        \justifying
        \textbf{More deaths flagged, more false positives.} Macro-F1 is
        similar to Claude few-shot, while sensitivity rises by
        \textbf{45.3 points} and accuracy falls by \textbf{18.1 points}.
      \end{beamercolorbox}

      \vspace{0.28em}
      \justifying
      The mixed system's full-test AUROC is \textbf{.706}, versus \textbf{.763}
      for a single GPT-OSS agent. The benefit is at the default cutoff,
      rather than stronger ranking. Task-supervised KARE has higher
      macro-F1 (90.4).
    \end{block}

    \vspace{0.27cm}

    \begin{block}{Integrator Swap Across All Test Sets}
      \centering
      {\fontsize{20}{24}\selectfont Qwen instead of GPT-OSS at integration;
      analysts and retrieval model fixed}\par\vspace{0.22em}
      {\fontsize{21}{25}\selectfont
      \begin{tabular}{@{}lrr@{}}
        \toprule
        \textbf{Outcome} & \textbf{$\Delta$ Sens.} & \textbf{$\Delta$ Macro-F1} \\
        \midrule
        \rowcolor{IntegratorGold!24}
        MIMIC-IV mortality & +46.4 & +4.4 \\
        MIMIC-III mortality & +37.1 & $-$16.7 \\
        MIMIC-IV readmission & +51.0 & $-$12.5 \\
        MIMIC-III readmission & +54.1 & $-$12.6 \\
        \bottomrule
      \end{tabular}}

      \vspace{0.26em}
      \justifying
      Changes are \textbf{percentage points} versus the all-large pipeline.
      Sensitivity rises in all four settings, but macro-F1 improves only on
      MIMIC-IV mortality.
    \end{block}
  \end{column}

  \begin{column}{0.315\textwidth}
    \begin{block}{Mechanism: Where Risk Crosses 0.5}
      \centering
      \includegraphics[width=0.80\linewidth]
        {Newposter-PromptLink/figures/mortality_condition_count.png}

      \vspace{0.12em}
      \justifying
      \textbf{Figure 2.} Among \emph{survivors}, both integrators raise
      predicted risk as recorded conditions increase. Qwen crosses the
      0.5 decision threshold earlier, creating more false positives.
      Error bars show standard error. The mixed system also flags
      \textbf{56\%} of decedents missed by the all-large system.
    \end{block}

    \vspace{0.27cm}

    \begin{block}{Boundary: Readmission}
      \begin{beamercolorbox}[wd=\linewidth,rounded=true,sep=0.45em]{boundarybox}
        \centering
        \textbf{MIMIC-IV, all-large $\rightarrow$ mixed}\par\vspace{0.16em}
        Sensitivity \textbf{43.5 $\rightarrow$ 94.5}\quad
        Macro-F1 \textbf{47.2 $\rightarrow$ 34.7}
      \end{beamercolorbox}
      \vspace{0.22em}
      \justifying
      The lower threshold gives about \textbf{4\% specificity}. A
      bag-of-codes probe reaches AUROC \textbf{.77} for mortality but only
      \textbf{.59} for readmission on MIMIC-IV: the available record has
      weaker signal for this post-discharge outcome.
    \end{block}
  \end{column}
